hokand_events.1 = {
	type = country_event
	placement = root
	duration = 3
	title = hokand_events.1.t
	desc = hokand_events.1.d
	flavor = hokand_events.1.f
	event_image = {
		video = "ep1_persian_court"
	}
	on_created_soundeffect = "event_default"
	immediate = {
		if = {
			limit = {
				exists = heir
			}
			heir = {
				kill_character = yes
			}
		}
		if = {
			limit = {
				exists = ruler
			}
			ruler = {
				kill_character = yes
			}
		}
		create_character = {
			template = KOK_murad_beg
			save_scope_as = murad_beg_target
		}
	}
	option = {
		name = hokand_events.1.a
		default_option = yes
		scope:murad_beg_target ?= {
			set_character_as_ruler = yes
		}
		add_journal_entry = {
			type = je_hokand_void_of_authority
		}
	}
}

hokand_events.2 = {
	type = country_event
	placement = root
	duration = 3
	title = hokand_events.2.t
	desc = hokand_events.2.d
	flavor = hokand_events.2.f
	event_image = {
		video = "ep1_transfer_of_authority"
	}
	on_created_soundeffect = "event_default"
	immediate = {
		# Heir must die BEFORE ruler: killing the ruler first promotes the heir
		# via succession, so a later `exists = heir` / heir kill never fires.
		if = {
			limit = {
				exists = heir
			}
			heir = {
				kill_character = yes
			}
		}
		if = {
			limit = {
				exists = ruler
			}
			ruler = {
				kill_character = yes
			}
		}
		create_character = {
			template = YTS_yakub_beg
			is_general = yes
			save_scope_as = yakub_beg_target
			on_created = {
				set_commander_rank = 6
			}
		}
	}
	option = {
		name = hokand_events.2.a
		default_option = yes
		scope:yakub_beg_target ?= {
			set_character_as_ruler = yes
		}
		
		custom_tooltip = {
			text = hokand_yakub_radicals_reduced_tt
			add_radicals = {
				value = -0.9
				culture = cu:uzbek
			}
		}
		add_loyalists = {
			value = 0.9
			culture = cu:uzbek
		}
		add_radicals = {
			value = 0.2
			culture = cu:kirgiz
		}
		add_primary_culture = cu:uzbek
		remove_primary_culture = cu:kirgiz
		set_capital = STATE_UZBEKIA
		# East Turkestan homeland claims granted when Liberation JE opens (not on complete).
		# Yakub Beg rule JE is added later in hokand_events.3 (tag → YTS).
		# Unlock before add so JE possible/show stay true even if ruler template
		# is not yet readable on the same tick.
		set_variable = kashgar_rebellion_unlocked
		add_journal_entry = {
			type = je_hokand_secure_authority
		}
		add_journal_entry = {
			type = je_kashgar_rebellion
		}
		custom_tooltip = {
			text = kashgar_rebellion_east_claims_tt
			s:STATE_KIRGHIZIA = { add_claim = root }
			s:STATE_TIANSHAN = { add_claim = root }
			s:STATE_DZUNGARIA = { add_claim = root }
			s:STATE_JETISY = { add_claim = root }
		}
		c:GBR = {
			add_involvement = {
				strategic_region = sr:region_central_asia
				value = 3000
			}
		}
		if = {
			limit = { exists = c:TUR }
			c:TUR = {
				add_involvement = {
					strategic_region = sr:region_central_asia
					value = 3000
				}
			}
		}
		add_involvement = {
			strategic_region = sr:region_near_east
			value = 3000
		}
		custom_tooltip = {
			text = hokand_yakub_cavalry_tt
			hidden_effect = {
				create_military_formation = {
					type = army
					hq_region = root.capital.region
					name = yakub_beg_cavalry
					save_scope_as = yakub_cavalry_formation
					combat_unit = {
						type = unit_type:combat_unit_type_hussars
						state_region = root.capital.state_region
						count = 20
					}
				}
				scope:yakub_beg_target = {
					add_character_role = general
					set_commander_rank = 6
					transfer_to_formation = scope:yakub_cavalry_formation
				}
			}
		}
	}
}
