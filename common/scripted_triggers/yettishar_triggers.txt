	}

# Full Harezm control — entire STATE_KHIVA, not a split-state share with TRM.
yettishar_turkmen_owns_full_harezm = {
	owns_entire_state_region = STATE_KHIVA
	OR = {
		NOT = { exists = c:TRM }
		c:TRM = {
			NOT = { has_state_in_state_region = STATE_KHIVA }
		}
	}
}

# Contested Harezm secured: we hold KHIVA and TRM no longer holds any share.
# Does NOT pass on a pre-existing YTS/TRH split share while TRM still owns part.
yettishar_turkmen_contested_harezm_secured = {
	OR = {
		yettishar_turkmen_owns_full_harezm = yes
		AND = {
			exists = c:TRM
			has_state_in_state_region = STATE_KHIVA
			c:TRM = {
				NOT = { has_state_in_state_region = STATE_KHIVA }
			}
		}
	}
}

# Contested Uralsk secured: we hold a share of STATE_URALSK and TRM holds none.
# Does NOT require entire region (RUS/KZH also hold provinces).
yettishar_turkmen_contested_uralsk_secured = {
	has_state_in_state_region = STATE_URALSK
	OR = {
		NOT = { exists = c:TRM }
		c:TRM = {
			NOT = { has_state_in_state_region = STATE_URALSK }
		}
	}
}

# Both colonization objectives: contested Harezm + contested Uralsk.

yettishar_turkmen_colonization_objectives_secured = {

	yettishar_turkmen_contested_uralsk_secured = yes

# Mongol PU subject still under root (not annexed / not self).

	custom_tooltip = {
		text = yettishar_mongol_is_pu_subject_tt
		exists = c:MGL
		NOT = { c:MGL ?= THIS }
		c:MGL = {
			is_subject_of = root
			is_subject_type = subject_type_personal_union
		}
	}
}

# Colonization war button: root or any subject borders TRM / Turkmenia / Merz.
# Raw OR — engine hoverable neighbour/owner triggers (no flat custom_tooltip wrapper).
yettishar_borders_turkmen_lands = {
	OR = {
		any_neighbouring_state = {
			OR = {
				AND = {
					exists = c:TRM
					owner = c:TRM
				}
				state_region = s:STATE_TURKMENIA
				state_region = s:STATE_MERZ
			}
		}
		any_subject_or_below = {
			any_neighbouring_state = {
				OR = {
					AND = {
						exists = c:TRM
						owner = c:TRM
					}
					state_region = s:STATE_TURKMENIA
					state_region = s:STATE_MERZ
				}
			}
		}
	}
}
