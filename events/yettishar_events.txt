# ===========================================================================
# 210. Expel Russians (Greater Turkestan) — retaliation via railways
# Order: create_pop tatar 15%
#        → move non-islamic russian (leave 30% of original)
#          across Ingria/Moscow/Kiev (capital fallback if none)
#        → create_mass_migration: russian → Moscow (else Ingria, else Kiev)
#          Ingria covered by 3-way instant move only
#        → 12-month push on origin states + pull on dests (soft leftover wave)
#        → SoL modifier + RUS −100 (no culture acceptance penalty)
# No kill step: kill_population_percent_in_state has no religion filter (would
# hit islamic Russians); every_scope_pop + kill_population_in_state had the
# total_size*ratio → 0 bug. Move-only guarantees Muslims survive and exact
# 30% non-islamic remain. Sequential move_partial_pop compounds: for N dest,
# each r satisfies (1-r)^N = 0.30 → 3:0.331 / 2:0.452 / 1:0.70
# Mass migration is culture-wide (no religion arg) — soft leftover OK.
# ===========================================================================
yettishar_events.210 = {
	type = country_event
	placement = root
	duration = 3
	title = yettishar_events.210.t
	desc = yettishar_events.210.d
	flavor = yettishar_events.210.f
	event_image = {
		video = "ep1_central_asian_city"
	}
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	on_created_soundeffect = "event_default"
	option = {
		name = yettishar_events.210.a
		default_option = yes
		custom_tooltip = {
			text = yettishar_events.210.a_tt
			hidden_effect = {
				if = {
					limit = { exists = c:RUS }
					c:RUS = {
						random_scope_state = {
							limit = { state_region = s:STATE_INGRIA }
							save_scope_as = ingria_dest
						}
						random_scope_state = {
							limit = { state_region = s:STATE_MOSCOW }
							save_scope_as = moscow_dest
						}
						random_scope_state = {
							limit = { state_region = s:STATE_KIEV }
							save_scope_as = kiev_dest
						}
					}
					# Capital fallback only when none of the three exist
					if = {
						limit = {
							NOT = { exists = scope:ingria_dest }
							NOT = { exists = scope:moscow_dest }
							NOT = { exists = scope:kiev_dest }
						}
						c:RUS = {
							if = {
								limit = { exists = capital }
								capital = { save_scope_as = capital_exile_dest }
							}
						}
					}
				}
				every_scope_state = {
					limit = {
						any_scope_pop = {
							culture = cu:russian
						}
					}
					# 1) Tatar settlement first (pre-expulsion population base)
					create_pop = {
						culture = tatar
						religion = sunni
						size = {
							value = this.state_population
							multiply = 0.15
						}
					}
					# 2) Move non-islamic Russian — leave 30% of original.
					#    Compound ratios: (1-r)^N = 0.30
					yettishar_expel_non_islamic_culture_pops = { CULTURE = cu:russian }
				}
				# Mass migration: one create (same-culture multi-target overwrites).
				# russian → Moscow preferred; no eligibility gate.
				if = {
					limit = { exists = scope:moscow_dest }
					scope:moscow_dest = {
						create_mass_migration = { origin = root culture = cu:russian }
					}
				}
				else_if = {
					limit = { exists = scope:ingria_dest }
					scope:ingria_dest = {
						create_mass_migration = { origin = root culture = cu:russian }
					}
				}
				else_if = {
					limit = { exists = scope:kiev_dest }
					scope:kiev_dest = {
						create_mass_migration = { origin = root culture = cu:russian }
					}
				}
				# Fallback mass mig only when none of the three exist
				if = {
					limit = {
						NOT = { exists = scope:ingria_dest }
						NOT = { exists = scope:moscow_dest }
						NOT = { exists = scope:kiev_dest }
						exists = c:RUS
					}
					c:RUS = {
						if = {
							limit = { exists = capital }
							capital = {
								create_mass_migration = { origin = root culture = cu:russian }
							}
						}
						else = {
							ordered_scope_state = {
								order_by = migration_pull
								position = 0
								check_range_bounds = no
								create_mass_migration = { origin = root culture = cu:russian }
							}
						}
					}
				}
				# Soft wave: push leftover russian out; pull into exile hubs (12 months)
				every_scope_state = {
					limit = {
						any_scope_pop = {
							culture = cu:russian
						}
					}
					add_modifier = {
						name = modifier_gt_expulsion_push
						months = 12
					}
				}
				if = {
					limit = { exists = scope:ingria_dest }
					scope:ingria_dest = {
						add_modifier = {
							name = modifier_gt_exile_pull
							months = 12
						}
					}
				}
				if = {
					limit = { exists = scope:moscow_dest }
					scope:moscow_dest = {
						add_modifier = {
							name = modifier_gt_exile_pull
							months = 12
						}
					}
				}
				if = {
					limit = { exists = scope:kiev_dest }
					scope:kiev_dest = {
						add_modifier = {
							name = modifier_gt_exile_pull
							months = 12
						}
					}
				}
				if = {
					limit = {
						NOT = { exists = scope:ingria_dest }
						NOT = { exists = scope:moscow_dest }
						NOT = { exists = scope:kiev_dest }
						exists = scope:capital_exile_dest
					}
					scope:capital_exile_dest = {
						add_modifier = {
							name = modifier_gt_exile_pull
							months = 12
						}
					}
				}
				set_variable = yettishar_gt_russians_expelled
				# Culture acceptance must use add_culture_acceptance_modifier
				# (country_*_cultural_acceptance_add in static modifiers does not apply).
				add_culture_acceptance_modifier = {
					culture = cu:russian
					multiplier = -15
				}
				add_culture_acceptance_modifier = {
					culture = cu:ukrainian
					multiplier = -15
				}
				add_modifier = {
					name = modifier_gt_russian_unwelcome
				}
				if = {
					limit = { exists = c:RUS }
					c:RUS = {
						change_relations = {
							country = root
							value = -100
						}
					}
				}
			}
		}
	}
}
