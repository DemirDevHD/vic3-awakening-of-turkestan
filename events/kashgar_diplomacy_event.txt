# 6. Cin Kabul Etti (Bariscil Transfer)
# ===========================================================================
kashgar_diplomacy_events.6 = {
	type = country_event
	placement = root
	title = kashgar_diplomacy_events.6.t
	desc = kashgar_diplomacy_events.6.d
	flavor = kashgar_diplomacy_events.6.f
	event_image = { video = "ep1_redcoats" }
	icon = "gfx/interface/icons/company_icons/historical_company_icons/gb_eic.dds"
	trigger = { always = yes }
	immediate = {
		# 1. KIRGIZİSTAN (s:STATE_KIRGHIZIA) -> DOĞRUDAN ROOT (HOKAND / OYUNCU)
		s:STATE_KIRGHIZIA = {
			every_scope_state = {
				limit = { owner = c:CHI }
				set_state_owner = root
				set_state_type = incorporated
			}
		}

		# 2. KAŞGAR EMİRLİĞİ / UYGURİSTAN (c:XIN) PU
		# Mode decided BEFORE create_country (seeding last CHI share must not
		# flip into release and steal player Jetisy):
		#   Liberation — CHI/XBR still holds ET: seed+fill from CHI/XBR only
		#   Release — no CHI/XBR ET left: seed+fill from root non-capital → XIN PU
		if = {
			limit = {
				AND = { exists = c:XIN c:XIN = root }
			}
			s:STATE_TIANSHAN = {
				every_scope_state = {
					limit = {
						owner = c:CHI
					}
					set_state_owner = root
					set_state_type = incorporated
				}
			}
			s:STATE_DZUNGARIA = {
				every_scope_state = {
					limit = {
						owner = c:CHI
					}
					set_state_owner = root
					set_state_type = incorporated
				}
			}
			s:STATE_JETISY = {
				every_scope_state = {
					limit = {
						owner = c:CHI
					}
					set_state_owner = root
					set_state_type = incorporated
				}
			}
		}
		else_if = {
			# Liberation: Qing / XBR still holds East Turkestan homeland
			limit = {
				OR = {
					s:STATE_TIANSHAN = {
						any_scope_state = {
							owner = c:CHI
						}
					}
					s:STATE_DZUNGARIA = {
						any_scope_state = {
							owner = c:CHI
						}
					}
					s:STATE_JETISY = {
						any_scope_state = {
							owner = c:CHI
						}
					}
				}
			}
			if = {
				limit = {
					NOT = { exists = c:XIN }
				}
				s:STATE_TIANSHAN = {
					random_scope_state = {
						limit = {
							owner = c:CHI
						}
						create_country = {
							tag = XIN
							origin = this.owner
							state = this
							on_created = {
								set_country_type = unrecognized
								activate_law = law_type:law_monarchy
								activate_law = law_type:law_autocracy
								regime_change = yes
								set_capital = STATE_TIANSHAN
							}
						}
					}
				}
				if = {
					limit = { NOT = { exists = c:XIN } }
					s:STATE_DZUNGARIA = {
						random_scope_state = {
							limit = {
								owner = c:CHI
							}
							create_country = {
								tag = XIN
								origin = this.owner
								state = this
								on_created = {
									set_country_type = unrecognized
									activate_law = law_type:law_monarchy
									activate_law = law_type:law_autocracy
									regime_change = yes
									set_capital = STATE_DZUNGARIA
								}
							}
						}
					}
				}
				if = {
					limit = { NOT = { exists = c:XIN } }
					s:STATE_JETISY = {
						random_scope_state = {
							limit = {
								owner = c:CHI
							}
							create_country = {
								tag = XIN
								origin = this.owner
								state = this
								on_created = {
									set_country_type = unrecognized
									activate_law = law_type:law_monarchy
									activate_law = law_type:law_autocracy
									regime_change = yes
									set_capital = STATE_JETISY
								}
							}
						}
					}
				}
			}
			if = {
				limit = { exists = c:XIN }
				s:STATE_TIANSHAN = {
					every_scope_state = {
						limit = {
							owner = c:CHI
						}
						set_state_owner = c:XIN
						set_state_type = incorporated
					}
				}
				s:STATE_DZUNGARIA = {
					every_scope_state = {
						limit = {
							owner = c:CHI
						}
						set_state_owner = c:XIN
						set_state_type = incorporated
					}
				}
				s:STATE_JETISY = {
					every_scope_state = {
						limit = {
							owner = c:CHI
						}
						set_state_owner = c:XIN
						set_state_type = incorporated
					}
				}
				if = {
					limit = {
						OR = {
							c:KOK = root
							c:YTS = root
						}
					}
					if = {
						limit = {
							c:XIN = { is_subject_of = root }
						}
						c:XIN = {
							change_subject_type = subject_type_personal_union
						}
					}
					else = {
						create_diplomatic_pact = {
							country = c:XIN
							type = personal_union
						}
					}
				}
			}
			else = {
				# Founding failed — Qing leftovers to root
				s:STATE_TIANSHAN = {
					every_scope_state = {
						limit = {
							owner = c:CHI
						}
						set_state_owner = root
						set_state_type = incorporated
					}
				}
				s:STATE_DZUNGARIA = {
					every_scope_state = {
						limit = {
							owner = c:CHI
						}
						set_state_owner = root
						set_state_type = incorporated
					}
				}
				s:STATE_JETISY = {
					every_scope_state = {
						limit = {
							owner = c:CHI
						}
						set_state_owner = root
						set_state_type = incorporated
					}
				}
			}
		}
		else = {
			# Release: no CHI/XBR ET left — found XIN from player homeland as PU
			if = {
				limit = {
					NOT = { exists = c:XIN }
				}
				s:STATE_TIANSHAN = {
					random_scope_state = {
						limit = {
							owner = root
							NOT = { is_capital = yes }
						}
						create_country = {
							tag = XIN
							origin = this.owner
							state = this
							on_created = {
								set_country_type = unrecognized
								activate_law = law_type:law_monarchy
								activate_law = law_type:law_autocracy
								regime_change = yes
								set_capital = STATE_TIANSHAN
							}
						}
					}
				}
				if = {
					limit = { NOT = { exists = c:XIN } }
					s:STATE_DZUNGARIA = {
						random_scope_state = {
							limit = {
								owner = root
								NOT = { is_capital = yes }
							}
							create_country = {
								tag = XIN
								origin = this.owner
								state = this
								on_created = {
									set_country_type = unrecognized
									activate_law = law_type:law_monarchy
									activate_law = law_type:law_autocracy
									regime_change = yes
									set_capital = STATE_DZUNGARIA
								}
							}
						}
					}
				}
				if = {
					limit = { NOT = { exists = c:XIN } }
					s:STATE_JETISY = {
						random_scope_state = {
							limit = {
								owner = root
								NOT = { is_capital = yes }
							}
							create_country = {
								tag = XIN
								origin = this.owner
								state = this
								on_created = {
									set_country_type = unrecognized
									activate_law = law_type:law_monarchy
									activate_law = law_type:law_autocracy
									regime_change = yes
									set_capital = STATE_JETISY
								}
							}
						}
					}
				}
			}
			if = {
				limit = { exists = c:XIN }
				s:STATE_TIANSHAN = {
					every_scope_state = {
						limit = {
							owner = root
							NOT = { is_capital = yes }
						}
						set_state_owner = c:XIN
						set_state_type = incorporated
					}
				}
				s:STATE_DZUNGARIA = {
					every_scope_state = {
						limit = {
							owner = root
							NOT = { is_capital = yes }
						}
						set_state_owner = c:XIN
						set_state_type = incorporated
					}
				}
				s:STATE_JETISY = {
					every_scope_state = {
						limit = {
							owner = root
							NOT = { is_capital = yes }
						}
						set_state_owner = c:XIN
						set_state_type = incorporated
					}
				}
				if = {
					limit = {
						OR = {
							c:KOK = root
							c:YTS = root
						}
					}
					if = {
						limit = {
							c:XIN = { is_subject_of = root }
						}
						c:XIN = {
							change_subject_type = subject_type_personal_union
						}
					}
					else = {
						create_diplomatic_pact = {
							country = c:XIN
							type = personal_union
						}
					}
				}
			}
		}

		# 2. Destekçi Ülkelerin (Kabul Edenlerin) Ödülleri
		if = {
			limit = {
				has_variable = kashgar_has_russian_support
				exists = c:RUS
			}
			s:STATE_AMUR = {
				every_scope_state = {
					limit = { owner = c:CHI }
					set_state_owner = c:RUS
				}
			}
			s:STATE_OUTER_MANCHURIA = {
				every_scope_state = {
					limit = { owner = c:CHI }
					set_state_owner = c:RUS
				}
			}
			s:STATE_SAKHALIN = {
				every_scope_state = {
					limit = { owner = c:CHI }
					set_state_owner = c:RUS
				}
			}
		}
		if = {
			limit = {
				has_variable = kashgar_russian_neutral
				exists = c:RUS
			}
			root = { set_owes_obligation_to = { country = c:RUS setting = yes } }
		}
		if = {
			limit = {
				has_variable = kashgar_has_british_support
				exists = c:GBR
			}
			create_treaty = {
				first_country = root
				second_country = c:GBR
				is_draft = no
				binding_period = { years = 20 }
				articles_to_create = {
					{
						article = trade_privilege
						source_country = root
						target_country = c:GBR
					}
				}
			}
			if = {
				limit = { exists = c:CHI }
				create_treaty = {
					first_country = c:CHI
					second_country = c:GBR
					is_draft = no
					binding_period = { years = 20 }
					articles_to_create = {
						{
							article = trade_privilege
							source_country = c:CHI
							target_country = c:GBR
						}
					}
				}
			}
			c:GBR = { add_treasury = 25000 }
		}
		if = {
			limit = {
				has_variable = kashgar_has_ottoman_support
				exists = c:TUR
			}
			c:TUR = {
				add_modifier = {
					name = modifier_kashgar_caliphate_prestige
					months = 240
				}
				add_treasury = 25000
			}
		}

		# 3. Ateşkes, Hazineler ve Geçici İlişkilerin Sıfırlanması
		create_bidirectional_truce = { country = c:CHI months = 60 }
		add_treasury = 50000
		set_variable = kashgar_crisis_won
		set_variable = kashgar_victory_event_fired

		if = {
			limit = { exists = c:RUS NOT = { c:RUS = root } c:RUS = { has_variable = kashgar_temp_relations_locked } }
			c:RUS = {
				change_relations = { country = root value = -60 }
				if = { limit = { exists = c:CHI NOT = { c:CHI = root } } change_relations = { country = c:CHI value = 60 } }
				remove_variable = kashgar_temp_relations_locked
			}
		}
		if = {
			limit = { exists = c:GBR NOT = { c:GBR = root } c:GBR = { has_variable = kashgar_temp_relations_locked } }
			c:GBR = {
				change_relations = { country = root value = -60 }
				if = { limit = { exists = c:CHI NOT = { c:CHI = root } } change_relations = { country = c:CHI value = 60 } }
				remove_variable = kashgar_temp_relations_locked
			}
		}
		if = {
			limit = { exists = c:TUR NOT = { c:TUR = root } c:TUR = { has_variable = kashgar_temp_relations_locked } }
			c:TUR = {
				change_relations = { country = root value = -60 }
				if = { limit = { exists = c:CHI NOT = { c:CHI = root } } change_relations = { country = c:CHI value = 60 } }
				remove_variable = kashgar_temp_relations_locked
			}
		}
		if = {
			limit = { exists = c:KOK NOT = { c:KOK = root } c:KOK = { has_variable = kashgar_temp_relations_locked } }
			c:KOK = {
				change_relations = { country = root value = -60 }
				if = { limit = { exists = c:CHI NOT = { c:CHI = root } } change_relations = { country = c:CHI value = 60 } }
				remove_variable = kashgar_temp_relations_locked
			}
		}
		if = {
			limit = { exists = c:RUS NOT = { c:RUS = root } c:RUS = { has_variable = kashgar_temp_neutral_locked } }
			c:RUS = {
				if = { limit = { exists = c:CHI NOT = { c:CHI = root } } change_relations = { country = c:CHI value = 80 } }
				remove_variable = kashgar_temp_neutral_locked
			}
		}
		if = {
			limit = { exists = c:YTS c:YTS = { has_variable = kashgar_temp_relations_locked } }
			c:YTS = { remove_variable = kashgar_temp_relations_locked }
		}
		if = {
			limit = { exists = c:XIN c:XIN = { has_variable = kashgar_temp_relations_locked } }
			c:XIN = { remove_variable = kashgar_temp_relations_locked }
		}
	}
	option = {
		name = kashgar_diplomacy_events.6.a
	}
}

	type = country_event

			change_relations = { country = c:CHI value = -80 }
			end_truce = c:CHI
			set_variable = kashgar_liberation_play_started

				}

					limit = {
						binds = c:CHI

					}
					withdraw = { country = PREV }
				}
			}
			root = { set_owes_obligation_to = { country = c:GBR setting = yes } }
		}
		# RUSYA
		if = {
			limit = {
				root = { has_variable = kashgar_has_russian_support }
				exists = c:RUS
			}
			c:RUS = {
				end_truce = c:CHI
				change_relations = { country = root value = 60 }
				if = { limit = { exists = c:CHI } change_relations = { country = c:CHI value = -60 } }
				set_variable = kashgar_temp_relations_locked
				every_scope_treaty = {
					limit = {
						binds = c:CHI
						any_scope_article = { has_type = alliance }
					}
					withdraw = { country = PREV }
				}
				every_scope_treaty = {
					limit = {
						binds = c:CHI
						any_scope_article = { has_type = defensive_pact }
					}
					withdraw = { country = PREV }
				}
			}
		}
		# OSMANLI
		if = {
			limit = {
				root = { has_variable = kashgar_has_ottoman_support }
				exists = c:TUR
			}
			c:TUR = {
				end_truce = c:CHI
				change_relations = { country = root value = 60 }
				if = { limit = { exists = c:CHI } change_relations = { country = c:CHI value = -60 } }
				set_variable = kashgar_temp_relations_locked
				every_scope_treaty = {
					limit = {
						binds = c:CHI
						any_scope_article = { has_type = alliance }
					}
					withdraw = { country = PREV }
				}
				every_scope_treaty = {
					limit = {
						binds = c:CHI
						any_scope_article = { has_type = defensive_pact }
					}
					withdraw = { country = PREV }
				}
			}
		}
		# RUSYA TARAFSIZ - Ateşkes
		if = {
			limit = {
				root = { has_variable = kashgar_russian_neutral }
				exists = c:RUS
			}
			c:RUS = {
				create_bidirectional_truce = { country = root months = 12 }
				if = {
					limit = { exists = c:CHI }

						}
						withdraw = { country = PREV }
					}

					}
				}

					limit = { is_diplomatic_play_type = dp_yts_liberation }
					# Çin tarafındaysa yank; zaten initiator ally ise noop.
					yts_ensure_initiator_backer = { COUNTRY = root }
					if = {
						limit = {
							root = { any_subject_or_below = { exists = this } }
						}
						root = {
							every_subject_or_below = {
								end_truce = c:CHI
								prev.prev = {
									yts_ensure_initiator_backer = { COUNTRY = prev }
								}
							}
						}
					}
					if = {
						limit = { exists = scope:rus_target_amur }
						add_war_goal = {
							type = conquer_state
							primary_demand = yes
							target_country = c:CHI
							target_state = scope:rus_target_amur
							holder = root
						}
					}
					if = {
						limit = { exists = scope:rus_target_outerm }
						add_war_goal = {
							type = conquer_state
							primary_demand = yes
							target_country = c:CHI
							target_state = scope:rus_target_outerm
							holder = root
						}
