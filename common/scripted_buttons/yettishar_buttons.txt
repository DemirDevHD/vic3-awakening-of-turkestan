btn_turkmen_colonization_war = {
	name = btn_turkmen_colonization_war_name
	desc = btn_turkmen_colonization_war_desc
	visible = {
		always = yes
	}
	possible = {
		has_institution = institution_colonial_affairs
		yettishar_borders_turkmen_lands = yes
		custom_tooltip = {
			text = cooldown_turkmen_colonization_war_tt
			NOT = { has_variable = cooldown_turkmen_colonization_war }
		}
		custom_tooltip = {
			text = yettishar_turkmen_war_not_won_tt
			NOT = { has_variable = yettishar_turkmen_colonization_war_won }
		}
		custom_tooltip = {
			text = yettishar_turkmen_play_not_active_tt
			NOT = { has_variable = yettishar_turkmen_colonization_play_active }
		}
	}
	# Selected only after a real war/button win — never from a pre-existing KHIVA split share
	selected = {
		has_variable = yettishar_turkmen_colonization_war_won
	}
	effect = {
		# Marker: button was pressed (required for ownership fallbacks)
		set_variable = yettishar_turkmen_colonization_war_started
		set_variable = yettishar_turkmen_colonization_button_used
		# Instant win ONLY if Harezm + Uralsk objectives already secured
		if = {
			limit = {
				yettishar_turkmen_already_owns_colonization_targets = yes
			}
			set_variable = yettishar_turkmen_colonization_war_won
			je:je_yettishar_turkmenistan ?= {
				add_progress = {
					name = yettishar_turkmen_meselesi_bar
					value = 40
				}
			}
		}
		else_if = {
			limit = { exists = c:TRM }
			# Primary: TRM's Harezm (STATE_KHIVA) share
			s:STATE_KHIVA = {
				random_scope_state = {
					limit = { owner = c:TRM }
					save_scope_as = harezm_target_state
					add_claim = root
				}
			}
			# Secondary: TRM's Uralsk (STATE_URALSK) share
			s:STATE_URALSK = {
				random_scope_state = {
					limit = { owner = c:TRM }
					save_scope_as = uralsk_target_state
					add_claim = root
				}
			}
			# Fallback primary: any TRM state if Harezm share is gone
			if = {
				limit = { NOT = { exists = scope:harezm_target_state } }
				c:TRM = {
					random_scope_state = {
						save_scope_as = harezm_target_state
						add_claim = root
					}
				}
			}
			if = {
				limit = { exists = scope:harezm_target_state }
				set_variable = yettishar_turkmen_colonization_play_active
				create_diplomatic_play = {
					name = turkmen_colonization_play
					target_country = c:TRM
					type = dp_turkmen_colonization
					target_state = scope:harezm_target_state
				}
				# Add Uralsk as second primary demand (2 Infamy) — Kashgar multi-wargoal pattern
				if = {
					limit = { exists = scope:uralsk_target_state }
					random_diplomatic_play = {
						limit = {
							is_diplomatic_play_type = dp_turkmen_colonization
						}
						add_war_goal = {
							holder = root
							type = turkmen_colonization_uralsk
							primary_demand = yes
							target_country = c:TRM
							target_state = scope:uralsk_target_state
						}
					}
				}
			}
			else = {
			}
		}
		else = {
			# No TRM — fight whoever holds Harezm (and Uralsk if needed)
			s:STATE_KHIVA = {
				random_scope_state = {
					limit = {
						NOT = { owner = root }
					}
					save_scope_as = harezm_target_state
					owner = { save_scope_as = harezm_target_country }
					add_claim = root
				}
			}
			s:STATE_URALSK = {
				random_scope_state = {
					limit = {
						NOT = { owner = root }
					}
					save_scope_as = uralsk_target_state
					add_claim = root
				}
			}
			if = {
				limit = {
					exists = scope:harezm_target_country
					exists = scope:harezm_target_state
				}
				set_variable = yettishar_turkmen_colonization_play_active
				create_diplomatic_play = {
					name = turkmen_colonization_play
					target_country = scope:harezm_target_country
					type = dp_turkmen_colonization
					target_state = scope:harezm_target_state
				}
				if = {
					limit = {
						exists = scope:uralsk_target_state
						scope:uralsk_target_state = {
							owner = scope:harezm_target_country
						}
					}
					random_diplomatic_play = {
						limit = {
							is_diplomatic_play_type = dp_turkmen_colonization
						}
						add_war_goal = {
							holder = root
							type = turkmen_colonization_uralsk
							primary_demand = yes
							target_country = scope:harezm_target_country
							target_state = scope:uralsk_target_state
						}
					}
				}
			}
			else_if = {
				limit = {
					yettishar_turkmen_colonization_objectives_secured = yes
				}
				set_variable = yettishar_turkmen_colonization_war_won
				je:je_yettishar_turkmenistan ?= {
					add_progress = {
						name = yettishar_turkmen_meselesi_bar
						value = 40
					}
				}
			}
			else = {
			}
		}
	}
	ai_chance = {
		value = 60
	}
}
