gov_yts_khanate = {
	transfer_of_power = hereditary

	male_ruler = "RULER_TITLE_EMIR"
	female_ruler = "RULER_TITLE_EMIRA"

	male_heir = "RULER_TITLE_PRINCE"
	female_heir = "RULER_TITLE_PRINCESS"

	possible = {
		has_law = law_type:law_monarchy
		OR = {
			AND = { exists = c:YTS c:YTS = ROOT }
			AND = { exists = c:TRH c:TRH = ROOT }
			AND = { exists = c:XIN c:XIN = ROOT }
		}
		NOT = { is_subject_type = subject_type_crown_land }
		has_gov_regency = no
		is_domain_alliance_gov = no
	}

	on_government_type_change = {
		change_to_hereditary = yes
	}
	on_post_government_type_change = {
		post_change_to_hereditary = yes
	}
}
