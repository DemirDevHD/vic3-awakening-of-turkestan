btn_hokand_grant_akmechet_command = {
    name = btn_hokand_grant_akmechet_command_name
    desc = btn_hokand_grant_akmechet_command_desc

    visible = {
        always = yes
    }

    possible = {
        authority >= 50
        custom_tooltip = {
            text = hokand_one_time_action_tt
            NOT = { has_variable = hokand_akmechet_granted }
        }
    }

    effect = {
        add_treasury = -1500
        je:je_hokand_void_of_authority ?= {
            add_progress = { name = hokand_yakub_support_bar value = 5 }
            add_progress = { name = hokand_authority_bar value = -5 }
        }
        add_modifier = {
            name = modifier_hokand_frontier_command
            months = 12
        }
        set_variable = hokand_akmechet_granted
    }
}

    desc = btn_hokand_secure_fergana_beys_desc

    visible = {
        always = yes
    }
    
    possible = {
        authority >= 100
        custom_tooltip = {
            text = hokand_one_time_action_tt
            NOT = { has_variable = hokand_fergana_beys_secured }
        }
        custom_tooltip = {
            text = hokand_fergana_bar_50_tt
            je:je_hokand_secure_authority ?= {
                "scripted_bar_progress(hokand_secure_fergana_bar)" >= 50
            }
        }
    }
    
    effect = {
        add_treasury = -5000
        set_variable = hokand_fergana_beys_secured
        trigger_event = { id = hokand_events.50 popup = yes }
    }
}
