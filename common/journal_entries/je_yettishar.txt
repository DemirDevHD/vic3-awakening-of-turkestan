					}

					limit = { exists = scope:trc_seed_state }
					create_country = {
						tag = TRC
						origin = root
						state = scope:trc_seed_state
					}
				}
			}
			# 3) Region-scoped transfer + engine auto-unify of split shares under TRC
			#    (chris transfer_state pattern). KHIVA stays with the player.
			#    Never transfers player capital.
			yettishar_trc_transfer_and_unify = {
				GIVER = root
			}
			# 3a) Also pull leftover shares from player subjects (not TRC itself)
			if = {
				limit = { exists = c:TRC }
				every_subject_or_below = {
					limit = { NOT = { this = c:TRC } }
					yettishar_trc_transfer_and_unify = {
						GIVER = this
					}
				}
			}
			# 3b) Capital MUST be Turkmenistan, never the Merv/Uralsk shares.
			#     STATE_MERZ and STATE_URALSK stay as separate regions under TRC —
			#     they are never merged into one state (vanilla set_capital pattern,
			#     see game/common/journal_entries/00_meiji_restoration.txt).
			if = {
				limit = {
					exists = c:TRC
					c:TRC = {
						any_scope_state = {
							state_region = s:STATE_TURKMENIA
						}
					}
				}
				c:TRC = {
					set_capital = STATE_TURKMENIA
				}
			}
			# 4) Personal union under player (YTS/TRH)
			if = {
				limit = { exists = c:TRC }
				if = {
					limit = {
						c:TRC = { is_subject_of = root }
					}
					c:TRC = {
						change_subject_type = subject_type_personal_union
					}
				}
				else = {
					create_diplomatic_pact = {
						country = c:TRC
						type = personal_union
					}
				}
			}
			set_variable = yettishar_turkmenistan_completed
		}
	}
	weight = 4000
	should_be_pinned_by_default_uninvolved_or_context = yes
