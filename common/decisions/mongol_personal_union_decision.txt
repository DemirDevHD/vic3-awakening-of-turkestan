# Mongol Personal Union — Bozkır Tahtını Kur / Establish the Steppe Throne
# SAFETY: never annex player; never give Gansu/Qinghai; keep >=1 non-whitelist state.
# PARSE: Vic3 decisions — NO icon; ai_chance uses value=.
# TOOLTIPS: raw engine triggers (like je_yettishar_modernization complete) so
# state regions nest; effect land list uses always=yes nested custom_tooltips.

mongol_personal_union_decision = {
	is_shown = {
		is_yts_or_trh = yes
		has_variable = kashgar_rebellion_finished
		NOT = { has_variable = yettishar_mongol_decision_taken }
	}

	possible = {
		is_yts_or_trh = yes
		custom_tooltip = {
			text = kashgar_rebellion_finished_tt
			has_variable = kashgar_rebellion_finished
		}
		custom_tooltip = {
			text = yettishar_mongol_decision_not_taken_tt
			NOT = { has_variable = yettishar_mongol_decision_taken }
		}
		# Raw — engine TRIGGER_HAS_STATE_IN_STATE_REGION with nestable STATE_REGION
		has_state_in_state_region = STATE_ULIASTAI
		has_state_in_state_region = STATE_URGA
		trigger_if = {
			limit = { NOT = { exists = c:MGL } }
			custom_tooltip = {
				text = mongol_pu_keeps_core_tt
				any_scope_state = {
					NOT = { yettishar_is_mongol_pu_transfer_state = yes }
				}
			}
			# Parent summary + raw OR children (no text=STATE_* overrides)
			custom_tooltip = {
				text = mongol_pu_has_seed_tt
				OR = {
					any_scope_state = {
						state_region = s:STATE_ALXA
						NOT = { is_capital = yes }
					}
					any_scope_state = {
						state_region = s:STATE_URGA
						NOT = { is_capital = yes }
					}
					any_scope_state = {
						state_region = s:STATE_ULIASTAI
						NOT = { is_capital = yes }
					}
					any_scope_state = {
						state_region = s:STATE_HINGGAN
						NOT = { is_capital = yes }
					}
					any_scope_state = {
						state_region = s:STATE_TRANS_BAIKAL
						NOT = { is_capital = yes }
					}
					any_scope_state = {
						state_region = s:STATE_BURYATIA
						NOT = { is_capital = yes }
					}
					any_scope_state = {
						state_region = s:STATE_IRKUTSK
						NOT = { is_capital = yes }
					}
					any_scope_state = {
						state_region = s:STATE_KALMYKIA
						NOT = { is_capital = yes }
					}
				}
			}
		}
	}

	when_taken = {
		custom_tooltip = mongol_pu_forms_union_tt
		# Effect-side nest: empty custom_tooltip { text = STATE_X } does not render;
		# always = yes makes each whitelist line appear under the summary.
		custom_tooltip = {
			text = mongol_pu_lands_summary_tt
			custom_tooltip = {
				text = STATE_ALXA
				always = yes
			}
			custom_tooltip = {
				text = STATE_URGA
				always = yes
			}
			custom_tooltip = {
				text = STATE_ULIASTAI
				always = yes
			}
			custom_tooltip = {
				text = STATE_HINGGAN
				always = yes
			}
			custom_tooltip = {
				text = STATE_TRANS_BAIKAL
				always = yes
			}
			custom_tooltip = {
				text = STATE_BURYATIA
				always = yes
			}
			custom_tooltip = {
				text = STATE_IRKUTSK
				always = yes
			}
			custom_tooltip = {
				text = STATE_KALMYKIA
				always = yes
			}
			custom_tooltip = {
				text = mongol_pu_no_gansu_qinghai_tt
				always = yes
			}
		}
		set_variable = yettishar_mongol_decision_taken
		add_modifier = {
			name = modifier_mongol_brotherhood_25
		}
		# Visible add_claim lines — engine nests state-region concepts
		s:STATE_TRANS_BAIKAL = { add_claim = root }
		s:STATE_BURYATIA = { add_claim = root }
		s:STATE_IRKUTSK = { add_claim = root }
		s:STATE_GANSU = { add_claim = root }
		s:STATE_SHENGJING = { add_claim = root }
		s:STATE_NORTHERN_MANCHURIA = { add_claim = root }
		s:STATE_SOUTHERN_MANCHURIA = { add_claim = root }
		s:STATE_OUTER_MANCHURIA = { add_claim = root }
		s:STATE_AMUR = { add_claim = root }
		hidden_effect = {
			yettishar_form_mongol_personal_union = {
				CAPITAL = STATE_ULIASTAI
			}
		}
		add_journal_entry = { type = je_mongol_steppe_unity }
	}

	ai_chance = {
		value = 80
	}
}
