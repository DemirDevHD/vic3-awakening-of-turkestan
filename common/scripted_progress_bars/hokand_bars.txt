hokand_authority_bar = {
    name = hokand_authority_bar_name
    desc = hokand_authority_bar_desc

    default_bad = yes # Han otoritesini sağladığı için bar yeşil renkli olur

    monthly_progress = {
        if = {
            limit = {
                legitimacy >= 50
            }
            add = {
                desc = "hokand_authority_from_legitimacy_tt"
                value = 5
            }
        }
        if = {
            limit = {
                is_at_war = yes
            }
            subtract = {
                desc = "hokand_authority_from_war_tt"
                value = 2
            }
        }
        if = {
            limit = {
                OR = {
                    has_variable = kashgar_victory_event_fired
                    has_variable = kashgar_crisis_won
                    has_state_in_state_region = s:STATE_TIANSHAN
                    any_subject_or_below = {
                        has_state_in_state_region = s:STATE_TIANSHAN
                    }
                }
            }
            add = {
                desc = "hokand_authority_from_kashgar_tt"
                value = 8
            }
        }
        if = {
            limit = {
                has_modifier = modifier_hokand_tribal_loyalty
            }
            add = {
                desc = "hokand_authority_from_tribes_tt"
                value = 2.5
            }
        }
        if = {
            limit = {
                legitimacy < 50
            }
            subtract = {
                desc = "hokand_authority_low_legitimacy_tt"
                value = 2
            }
        }
        if = {
            limit = {
                has_modifier = modifier_hokand_frontier_command
            }
            subtract = {
                desc = "hokand_authority_from_frontier_tt"
                value = 1
            }
        }
        if = {
            limit = {
                has_modifier = modifier_hokand_ulema_fatwa
            }
            subtract = {
                desc = "hokand_authority_from_fatwa_tt"
                value = 1
            }
        }
    }

    start_value = 300
    min_value = 0
    max_value = 450
}
