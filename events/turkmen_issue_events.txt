turkmen_issue_events.1 = {
	type = country_event
	placement = root
	duration = 3
	title = turkmen_issue_events.1.t
	desc = turkmen_issue_events.1.d
	flavor = turkmen_issue_events.1.f
	event_image = {
		video = "ep1_central_asian_city"
	}
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	on_created_soundeffect = "event_default"
	option = {
		name = turkmen_issue_events.1.a
		default_option = yes
		je:je_yettishar_turkmenistan ?= {
			add_progress = {
				name = yettishar_turkmen_meselesi_bar
				value = 18
			}
		}
		custom_tooltip = {
			text = turkmen_bar_gain_18_tt
		}
	}
	option = {
		name = turkmen_issue_events.1.b
		je:je_yettishar_turkmenistan ?= {
			add_progress = {
				name = yettishar_turkmen_meselesi_bar
				value = 10
			}
		}
		custom_tooltip = {
			text = turkmen_bar_gain_10_tt
		}
		random_scope_state = {
			limit = {
				OR = {
					state_region = s:STATE_TURKMENIA
					state_region = s:STATE_URALSK
					is_capital = yes
				}
			}
			add_radicals_in_state = {
				value = 0.03
			}
		}
	}
	option = {
		name = turkmen_issue_events.1.c
		add_radicals = {
			value = 0.05
			strata = lower
		}
		custom_tooltip = {
			text = turkmen_bar_gain_0_tt
		}
	}
}

	duration = 3

	desc = turkmen_issue_events.16.d
	flavor = turkmen_issue_events.16.f
	event_image = {
		video = "ep1_central_asian_city"
	}
	icon = "gfx/interface/icons/event_icons/event_industry.dds"
	on_created_soundeffect = "event_default"
	trigger = {
		OR = {
			c:YTS ?= THIS
			c:TRH ?= THIS
		}
		has_variable = yettishar_turkmenistan_started
		NOT = { has_variable = yettishar_turkmenistan_completed }
		has_journal_entry = je_yettishar_turkmenistan
	}
	option = {
		name = turkmen_issue_events.16.a
		default_option = yes
		if = {
			limit = {
				has_institution = institution_colonial_affairs
			}
			je:je_yettishar_turkmenistan ?= {
				add_progress = {
					name = yettishar_turkmen_meselesi_bar
					value = 12
				}
			}
			custom_tooltip = {
				text = turkmen_bar_gain_12_tt
			}
		}
		else = {
			je:je_yettishar_turkmenistan ?= {
				add_progress = {
					name = yettishar_turkmen_meselesi_bar
					value = 4
				}
			}
			custom_tooltip = {
				text = turkmen_bar_gain_4_tt
			}
			custom_tooltip = {
				text = turkmen_need_colonial_affairs_tt
			}
		}
	}
}
